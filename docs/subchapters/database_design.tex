	\caption{Tables description}
	\label{tab:tables_description}\\[-1.5ex]
	\hline
	Table name & Description \\
	\hline
	\endfirsthead
	user & A table containing information about users\\
	\hline
	user\_role & Join table for N-N relation\\
	\hline
	role & Table containing predefined set of roles\\
	\hline
	task & Table containing tasks to do\\
	\hline
	field & Table containing fields that have grapes planted on them\\
	\hline
	spraying & Table containing information about spraying done on fields\\
	\hline
	substance & Table containing user defined list of chemicals that can be used to spray\\
	\hline
	grape\_type & Table containing user defined list of grapes variety\\
	\hline
	equipment & Table containing information about winary and vineyard equipment e.g. fermentors, barrels \\
	\hline
	record & A table containing information about chemical parameter records\\
	\hline
	organisation & Table containing information about organisations\\
	\hline
	harvest & Tables containing amount of grapes, dates and on what field it was harvested\\
	\hline
	offering & Table containing list of wines that are available to buy\\
	\hline
	equipment\_type & Table containing predefined list of equipment typ e.g. fermentor, spraying machine\\
	\hline
	task\_type & Table containing pre-defined task type list(different task types have different fields)\\
	\hline
	grape\_color & Table containing predefined list of grape colours(red, white)\\
	\hline
	field\_harvest & Join table for N-N relation\\
	\hline
	must & A table containing information about a batch of must\\
	\hline
	must\_harvest & Join table for N-N relation\\
	\hline
	fermenting\_must & A table containing information about a batch of fermenting must\\
	\hline
	fermentation\_must & Join table for N-N relation\\
	\hline
	ingredient & A table containing information about ingredients\\
	\hline
	fermentation\_ingredient & Join table for N-N relation\\
	\hline
	bottled\_wine & A table containing information about a batch of  bottled wine\\
	\hline
	to\_be\_bottled & A table allowing for the use of different types of batches\\
	\hline
	aged\_wine & A table containing information about a batch of  aged wine\\
	\hline
	aged\_wine\_fermentation & Join table for N-N relation\\
	\hline
	blended\_wine & A table containing information about a batch of blended wine\\
	\hline
	blended\_wine\_ingredient & Join table for N-N relation\\
	\hline
	refresh\_token & Table containing refresh tokens\\
	\hline
	record\_type & A table containing information about record types\\
	\hline
	disease & Table containing stored pests attacks, fungal infections etc.\\
	\hline
\end{longtable}
